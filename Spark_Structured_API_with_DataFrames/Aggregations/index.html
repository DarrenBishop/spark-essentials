Aggregations & Grouping

val moviesDF = spark.read
  .option("inferSchema", "true")
  .json("path/to/movies.json")
// moviesDF: org.apache.spark.sql.package.DataFrame = [Creative_Type: string, Director: string ... 14 more fields]

Aggregations​

count (Count)​

moviesDF.select(count($"Major_Genre")).show()
// +------------------+
// |count(Major_Genre)|
// +------------------+
// |              2926|
// +------------------+
//

Count All​

moviesDF.select(count("*")).show()
// +--------+
// |count(1)|
// +--------+
// |    3201|
// +--------+
//

moviesDF.selectExpr("count(*)").show()

count_distinct (Count Distinct)​

moviesDF.select(count_distinct($"Major_Genre")).show()
// +---------------------------+
// |count(DISTINCT Major_Genre)|
// +---------------------------+
// |                         12|
// +---------------------------+
//

approx_count_distinct (Approximate Count)​

moviesDF.select(approx_count_distinct($"Major_Genre")).show()
// +----------------------------------+
// |approx_count_distinct(Major_Genre)|
// +----------------------------------+
// |                                12|
// +----------------------------------+
//

min & max​

moviesDF.select(min($"IMDB_Rating")).show()
// +----------------+
// |min(IMDB_Rating)|
// +----------------+
// |             1.4|
// +----------------+
// 
moviesDF.selectExpr("min(IMDB_Rating)").show()
// +----------------+
// |min(IMDB_Rating)|
// +----------------+
// |             1.4|
// +----------------+
//

sum​

moviesDF.select(sum($"US_Gross")).show()
// +-------------+
// |sum(US_Gross)|
// +-------------+
// | 140542660013|
// +-------------+
// 
moviesDF.selectExpr("sum(US_Gross)").show()
// +-------------+
// |sum(US_Gross)|
// +-------------+
// | 140542660013|
// +-------------+
//

avg (Average)​

moviesDF.select(avg($"Rotten_Tomatoes_Rating")).show()
// +---------------------------+
// |avg(Rotten_Tomatoes_Rating)|
// +---------------------------+
// |          54.33692373976734|

Grouping​

groupBy​

select count(*) from moviesDF group by Major_Genre
moviesDF
  .groupBy($"Major_Genre")
  .count()
  .show()
// +-------------------+-----+
// |        Major_Genre|count|
// +-------------------+-----+
// |          Adventure|  274|
// |               NULL|  275|
// |              Drama|  789|
// |        Documentary|   43|
// |       Black Comedy|   36|
// |  Thriller/Suspense|  239|
// |            Musical|   53|

// |Concert/Performance|    5|
// |             Horror|  219|

avg​

There are aggregation overloads e.g. avg that take String column names:

moviesDF
  .groupBy($"Major_Genre")
  .avg("IMDB_Rating")
  .show()
// +-------------------+------------------+
// |        Major_Genre|  avg(IMDB_Rating)|
// +-------------------+------------------+
// |          Adventure| 6.345019920318729|

agg​

Multiple aggregations can be specified on a group together using agg:

moviesDF
  .groupBy($"Major_Genre")
  .agg(
    "Movies" := count("*"),
    "Avg Rating" := avg("IMDB_Rating")
  )
  .orderBy($"Avg Rating")
  .show()
// +-------------------+------+------------------+
// |        Major_Genre|Movies|        Avg Rating|
// +-------------------+------+------------------+
// |             Horror|   219|5.6760765550239185|
// |             Comedy|   675| 5.853858267716529|
// |    Romantic Comedy|   137| 5.873076923076922|
// |             Action|   420| 6.114795918367349|
// |Concert/Performance|     5|             6.325|
// |          Adventure|   274| 6.345019920318729|
// |  Thriller/Suspense|   239| 6.360944206008582|
// |            Musical|    53|             6.448|
// |               NULL|   275|  6.50082644628099|
// |              Drama|   789| 6.773441734417339|
// |       Black Comedy|    36|6.8187500000000005|
// |            Western|    36| 6.842857142857142|
// |        Documentary|    43| 6.997297297297298|
